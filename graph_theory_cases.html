4b. MTSP：多旅行商与不同起终点

问题：有 K 名销售人员（或搜救人员、车辆），每人有各自的起点和终点。要求所有人合起来恰好访问所有客户节点一次，且任一人员的路径最“长”（用时/距离）尽可能短。

MTSP 数学表达

集合：V = 所有节点；S_k = 第 k 个人起点；E_k = 第 k 个人终点；I = V \ ∪_k {{S_k, E_k}}（中间节点）。

决策变量：

  • xijk ∈ {0,1}：第 k 个人是否从 i 直接到 j
  • T ≥ 0：所有人的最大行程（makespan）

目标函数：

min T

约束：

  • 每个中间节点恰好被一个人进入、一个人离开
  • 第 k 个人从 S_k 出发、到达 E_k
  • 每个人路径流量守恒
  • Σi,j cij · xijk ≤ T，∀k
  • 消除子回路（MTZ）
takeaway：把 TSP 的“最短总距离”改成“最短最长单线距离”，立刻对应到搜救、配送、巡逻中的公平与效率平衡。 
import pulp
import numpy as np
import matplotlib.pyplot as plt

# ===== 1. 数据输入 =====
# nodes: list[str]，所有节点的标签
nodes = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H']
# n: int，节点总数
n = len(nodes)

# coords: numpy.ndarray, shape (n, 2)，每个节点在二维平面上的随机坐标
np.random.seed(42)
coords = np.random.rand(n, 2) * 10

# dist: numpy.ndarray, shape (n, n)，对称距离矩阵；dist[i][j] 表示节点 i 到 j 的欧氏距离
dist = np.zeros((n, n))
for i in range(n):
    for j in range(n):
        dist[i][j] = np.linalg.norm(coords[i] - coords[j])

# salesmen: list[dict]，每名销售人员的元信息：姓名、起点、终点
salesmen = [
    {'name': 'S1', 'start': 'A', 'end': 'B'},
    {'name': 'S2', 'start': 'C', 'end': 'D'}
]
# K: int，销售人员数量
K = len(salesmen)

# start_ends: set[str]，所有起点和终点的集合，这些节点不需要被“访问”，只作为路径端点
start_ends = set(s['start'] for s in salesmen) | set(s['end'] for s in salesmen)
# intermediate: list[str]，需要被访问的中间节点（去掉所有起点/终点）
intermediate = [n for n in nodes if n not in start_ends]
# N: int，中间节点数量，用于 MTZ 子回路消除中的位置编号上界
N = len(intermediate)

# ===== 2. 创建 PuLP 模型 =====
# prob: pulp.LpProblem 实例，目标是最小化 makespan（所有人中最长的单线路径）
prob = pulp.LpProblem('MTSP_makespan', pulp.LpMinimize)

# ===== 3. 决策变量 =====
# x: dict[str, dict[str, dict[int, pulp.LpVariable]]]，三维二元变量 x[i][j][k]
#    x[i][j][k] == 1 表示第 k 名 salesman 的路径中包含从节点 i 直接到节点 j 的边
x = pulp.LpVariable.dicts('x', (nodes, nodes, range(K)), cat='Binary')

# T: pulp.LpVariable，连续变量，表示所有 salesmen 中单条路径长度的最大值（makespan）
T = pulp.LpVariable('T', lowBound=0)

# u: dict[str, dict[int, pulp.LpVariable]]，MTZ 辅助变量 u[i][k]
#    表示中间节点 i 在第 k 名 salesman 路径中的访问序号，范围 [1, N]
#    例如 u['E'][0] = 2 表示 E 是 S1 路径中第 2 个访问的中间节点
u = pulp.LpVariable.dicts('u', (intermediate, range(K)), lowBound=1, upBound=N, cat='Continuous')

# ===== 4. 目标函数 =====
# 最小化 makespan T，即让所有人中最长的那条路径尽可能短
prob += T

# ===== 5. 约束条件 =====
# 约束 1：禁止自环
# 含义：不允许从节点 i 直接走回 i，避免无意义的零长度边
for i in nodes:
    for k in range(K):
        prob += x[i][i][k] == 0

# 约束 2：每名 salesman 必须从自己的起点出发一次
# 含义：对第 k 个人，从 sk 出发的出边中只能选一条
for k, s in enumerate(salesmen):
    sk = s['start']
    prob += pulp.lpSum(x[sk][j][k] for j in nodes if j != sk) == 1

# 约束 3：每名 salesman 必须到达自己的终点一次
# 含义：对第 k 个人，进入 ek 的入边中只能选一条
for k, s in enumerate(salesmen):
    ek = s['end']
    prob += pulp.lpSum(x[i][ek][k] for i in nodes if i != ek) == 1

# 约束 4：salesman 不能进入自己的起点、也不能离开自己的终点
# 含义：起点只出不入，终点只入不出，保证路径方向是从起点到终点
for k, s in enumerate(salesmen):
    sk, ek = s['start'], s['end']
    prob += pulp.lpSum(x[i][sk][k] for i in nodes if i != sk) == 0
    prob += pulp.lpSum(x[ek][j][k] for j in nodes if j != ek) == 0

# 约束 5：每个中间节点恰好被一个人进入一次
# 含义：客户/中间节点不能被遗漏，也不能被两个人重复访问
for v in intermediate:
    prob += pulp.lpSum(x[i][v][k] for i in nodes if i != v for k in range(K)) == 1

# 约束 6：每个中间节点恰好被一个人离开一次
# 含义：与约束 5 配对，保证中间节点在路径中出现且仅出现一次
for v in intermediate:
    prob += pulp.lpSum(x[v][j][k] for j in nodes if j != v for k in range(K)) == 1

# 约束 7：每名 salesman 在每个中间节点流量守恒
# 含义：若第 k 个人进入了节点 v，则他也必须从 v 离开；保证路径连续不断开
for k in range(K):
    for v in intermediate:
        prob += (pulp.lpSum(x[i][v][k] for i in nodes if i != v)
                 == pulp.lpSum(x[v][j][k] for j in nodes if j != v))

# 约束 8：每名 salesman 的总路程不超过 makespan T
# 含义：T 必须大于等于每个人路径长度，因此最小化 T 就是最小化最长路径
for k in range(K):
    prob += T >= pulp.lpSum(
        dist[i][j] * x[nodes[i]][nodes[j]][k]
        for i in range(n) for j in range(n) if i != j
    )

# 约束 9：MTZ 子回路消除（每名 salesman 独立）
# 含义：若第 k 个人走了边 i -> j，则 j 在路径中的序号必须比 i 大；
#      这样任意两个被选中的节点之间都有明确的先后顺序，避免形成不连通的小圈。
# 数学形式：u_i^k - u_j^k + N * x_{ij}^k <= N - 1
for k in range(K):
    for i in intermediate:
        for j in intermediate:
            if i != j:
                prob += u[i][k] - u[j][k] + N * x[i][j][k] <= N - 1

# ===== 6. 求解 =====
prob.solve(pulp.PULP_CBC_CMD(msg=False, timeLimit=60))
print('Status:', pulp.LpStatus[prob.status])
print('Makespan T =', pulp.value(T))

# ===== 7. 从解中提取每个人的路径 =====
for k, s in enumerate(salesmen):
    # route: list[str]，第 k 名 salesman 按顺序访问的节点列表
    route = [s['start']]
    current = s['start']
    # 沿着 x[current][j][k] == 1 的边从起点走到终点
    while current != s['end']:
        nxt = None
        for j in nodes:
            if j != current and pulp.value(x[current][j][k]) > 0.5:
                nxt = j
                break
        if nxt is None:
            break
        route.append(nxt)
        current = nxt
    # cost: float，该 salesman 路径的总长度
    cost = sum(
        dist[nodes.index(route[i])][nodes.index(route[i+1])]
        for i in range(len(route) - 1)
    )
    print(f"{s['name']}: {' -> '.join(route)} = {cost:.2f}")

# ===== 8. 可视化 =====
plt.figure(figsize=(8, 6))
# colors: list[str]，每名 salesman 的路径颜色
colors = ['red', 'blue']
for k, s in enumerate(salesmen):
    # edges: list[tuple[str, str]]，第 k 名 salesman 被选中的所有边
    edges = [(i, j) for i in nodes for j in nodes
             if i != j and pulp.value(x[i][j][k]) > 0.5]
    for (u_node, v_node) in edges:
        ui, vi = nodes.index(u_node), nodes.index(v_node)
        plt.plot([coords[ui, 0], coords[vi, 0]], [coords[ui, 1], coords[vi, 1]],
                 color=colors[k], linewidth=2, alpha=0.7)
    # 绘制所有节点作为背景点
    plt.scatter(coords[:, 0], coords[:, 1], c='lightgray', s=300, zorder=3)
    # 标注节点标签；起点/终点用该 salesman 的颜色高亮
    for idx, node in enumerate(nodes):
        color = colors[k] if node in [s['start'], s['end']] else 'black'
        plt.text(coords[idx, 0], coords[idx, 1], node,
                 ha='center', va='center', fontsize=10,
                 color=color, fontweight='bold', zorder=4)
plt.title('MTSP: Minimize Maximum Tour Length')
plt.axis('off')
plt.show()

0. 案例导入：商场起火，为什么要用图论？

场景：周五晚上，商场一层的 Retail_A 店铺起火。烟不是只留在店里，而是沿着走廊、扶梯、楼梯井迅速扩散。作为应急响应人员，你需要回答：

  • 烟最快会从哪里蔓延？
  • 哪些通道应该立即关闭？
  • 顾客应该从哪个出口撤离？

建模思路：把商场内部的空间单元（店铺、走廊、楼梯）抽象成节点，把它们之间的连通路径抽象成边。于是“烟怎么扩散”“人怎么疏散”就变成了图上的传播与路径问题。

takeaway：图论不是画一张漂亮地图，而是把空间结构变成可计算对象。
import networkx as nx

# A minimal sanity check: a small abstract mall
G = nx.Graph()
G.add_nodes_from(['Entrance', 'Retail_A', 'Corridor', 'Stairwell'])
G.add_edges_from([
    ('Entrance', 'Retail_A'),
    ('Retail_A', 'Corridor'),
    ('Corridor', 'Stairwell')
])

print("Nodes (spatial units):", list(G.nodes()))
print("Edges (connections):", list(G.edges()))
print("Degree of each node:", dict(G.degree()))

启发式提问

Q1：面对商场火灾，你最关心的三个问题是什么？

参考方向：这三个问题通常对应模型的三个核心：火源/烟源位置、传播路径、人员疏散出口。

Q2（开放）：为什么用图论而不是只用一张平面图来分析？

回答 A：平面图只能看位置关系，图论能量化‘必须经过谁’‘距离多远’‘切断哪条边最有效’。

回答 B：图可以与传播算法、最短路径算法、中心性算法结合，实现自动模拟。

回答 C：图上的节点和边可以携带属性（长度、容量、通风效率），便于做敏感性分析。

回答 D：便于与优化工具结合，回答‘最优隔离门位置’这类设计问题。

Q3：在这个案例里，节点和边分别可以代表什么？

参考方向：节点代表空间单元：店铺、走廊、扶梯、楼梯间、设备间；边代表它们之间的连通路径，通常用长度、通风阻力等作为权重。

教师参考

本模块的核心方法是‘现实空间 → 图结构’。教师要让学生意识到：一个节点不是一个点，而是一个有属性的空间单元；一条边不是一条线，而是一个有方向/长度/容量的通道。先建立这种抽象，后面的传播模型才有意义。

1. 把商场抽象成图：节点、边与属性

这部分做什么：定义一个简化的商场通风网络。每个节点有三个属性：type（空间类型）、pos（平面位置）、state（初始状态）。边是无向的，表示烟雾/人员可以在两个方向流动。

我们根据节点坐标计算每条边的长度，因为长度会影响烟雾传播速率：走廊越长，烟越慢到达。

takeaway：给边加“长度”属性，是把几何信息注入图结构的第一步。
import networkx as nx
import matplotlib.pyplot as plt
import numpy as np

# Create an undirected graph for airflow/smoke propagation paths.
G_smoke = nx.Graph()

# Nodes: spatial units with type, position, and initial state.
nodes_data = {
    'Retail_A': {'type': 'store', 'pos': (0, 0)},   # fire origin
    'Retail_B': {'type': 'store', 'pos': (1, 0)},
    'Retail_C': {'type': 'store', 'pos': (0, -1)},
    'Corridor_1': {'type': 'corridor', 'pos': (0.5, 0.5)},
    'Corridor_2': {'type': 'corridor', 'pos': (0.5, -0.5)},
    'Escalator_Up': {'type': 'escalator', 'pos': (1.5, 0.5)},
    'Escalator_Down': {'type': 'escalator', 'pos': (1.5, -0.5)},
    'Stairwell_1': {'type': 'stairwell', 'pos': (2, 0)},
    'Equipment_Room': {'type': 'utility', 'pos': (-0.5, -0.5)}
}

for node_id, data in nodes_data.items():
    G_smoke.add_node(node_id, type=data['type'], pos=data['pos'], state='S')

# Edges: physical connections (undirected = smoke can travel both ways).
edges_data = [
    ('Retail_A', 'Corridor_1'),
    ('Corridor_1', 'Retail_B'),
    ('Retail_A', 'Corridor_2'),
    ('Corridor_2', 'Retail_C'),
    ('Retail_B', 'Escalator_Up'),
    ('Escalator_Up', 'Stairwell_1'),
    ('Retail_C', 'Escalator_Down'),
    ('Escalator_Down', 'Stairwell_1'),
    ('Corridor_2', 'Equipment_Room'),
    ('Stairwell_1', 'Retail_B'),
    ('Corridor_1', 'Stairwell_1')
]
G_smoke.add_edges_from(edges_data)

# Compute edge lengths from node positions.
pos = nx.get_node_attributes(G_smoke, 'pos')
for u, v in G_smoke.edges():

    G_smoke.edges[u, v]['length'] = length

# Visualize.
node_type_colors = {

node_colors = [node_type_colors[G_smoke.nodes[n]['type']] for n in G_smoke.nodes()]

plt.figure(figsize=(10, 8))
nx.draw_networkx_nodes(G_smoke, pos, node_color=node_colors, node_size=3000, alpha=0.9)
nx.draw_networkx_labels(G_smoke, pos, font_size=9, font_weight='bold')

nx.draw_networkx_edge_labels(G_smoke, pos, edge_labels=formatted, font_color='darkred', font_size=7)
plt.title("Mall Ventilation Network: Nodes and Edge Lengths")

plt.show()

print(f"Number of nodes: {G_smoke.number_of_nodes()}")
print(f"Number of edges: {G_smoke.number_of_edges()}")

启发式提问

Q1：为什么用无向图而不是有向图？

参考方向：烟在静止空气中会向各个方向扩散，人员在疏散前也可以在走廊里双向走动。只有当我们要专门模拟‘只能向下逃生的楼梯’或‘单向扶梯’时，才需要把边改成有向。

Q2：节点位置（pos）只影响可视化吗？

参考方向：不只是可视化。这里用它来计算边长度，而边长度会进入后面的传播概率公式——越长的通道，烟到达得越慢。

Q3（开放）：如果楼梯只能向下、电梯停运，有向图会怎么改？

回答 A：把 Stairwell 到上层的边保留为向下方向，移除向上的边。

回答 B：扶梯边可以删除，因为火灾时通常禁止使用。

回答 C：走廊在疏散阶段可能变成单向，避免对冲，需要把无向边拆成两条方向相反但有容量限制的有向边。

回答 D：有向图能表达疏散规则，但也会让某些节点变得‘不可达’，需要检查图的强连通性。

教师参考

本模块的解决方法是‘图抽象 + 属性标注’。教师要强调：节点属性（type, pos, state）和边属性（length）是后续所有计算的原料。让学生自己决定哪些属性重要，是建模训练的核心。

2. SIR 烟扩散模型：状态、转移与参数

为什么用 SIR？传染病模型和烟雾扩散有一个共同结构：某种‘东西’通过接触/邻接关系从一个单元传到另一个单元。我们可以借用 SIR 的框架：

  • S（Safe）：尚未被烟波及的区域。
  • I（Infected）：烟雾浓度超过安全阈值，处于危险中。
  • R（Recovered）：烟雾被排出或中和，区域重新安全。

转移规则：每个时间步，S 节点如果与 I 节点相邻，就以一定概率变成 I；I 节点以恢复率 γ 变成 R。边越长，传播概率越低。

import random
import networkx as nx
import matplotlib.pyplot as plt
import numpy as np

# Re-build the mall graph (self-contained for this block).
G_smoke = nx.Graph()
nodes_data = {

    'Retail_B': {'type': 'store', 'pos': (1, 0)},
    'Retail_C': {'type': 'store', 'pos': (0, -1)},
    'Corridor_1': {'type': 'corridor', 'pos': (0.5, 0.5)},
    'Corridor_2': {'type': 'corridor', 'pos': (0.5, -0.5)},
    'Escalator_Up': {'type': 'escalator', 'pos': (1.5, 0.5)},
    'Escalator_Down': {'type': 'escalator', 'pos': (1.5, -0.5)},
    'Stairwell_1': {'type': 'stairwell', 'pos': (2, 0)},

pos = nx.get_node_attributes(G_smoke, 'pos')
for u, v in G_smoke.edges():
    G_smoke.edges[u, v]['length'] = np.linalg.norm(np.array(pos[u]) - np.array(pos[v]))


def run_smoke_simulation(graph, fire_origin,
                         base_propagation_rate,  # base probability that smoke spreads from I to S per step
                         decay_factor,           # how much edge length reduces propagation rate
                         recovery_rate,          # base probability that I becomes R per step
                         intervention_boost,     # extra recovery rate when intervention is active
                         intervention_start):    # earliest time step intervention may activate
    """Run one stochastic SIR-style smoke simulation on the mall graph."""
    G_sim = graph.copy()
    for node in G_sim.nodes():
        G_sim.nodes[node]['state'] = 'S'
        G_sim.nodes[node]['infected_time'] = -1

    G_sim.nodes[fire_origin]['state'] = 'I'
    G_sim.nodes[fire_origin]['infected_time'] = 0

    intervention_active = False
    num_time_steps = 100
    network_states_over_time = []

    def counts(g):
        s = sum(1 for n in g.nodes() if g.nodes[n]['state'] == 'S')
        i = sum(1 for n in g.nodes() if g.nodes[n]['state'] == 'I')
        r = sum(1 for n in g.nodes() if g.nodes[n]['state'] == 'R')
        return s, i, r

    for t in range(num_time_steps):
        if t >= intervention_start and not intervention_active:
            if random.random() < 0.8:
                intervention_active = True

        new_states = {}
        for node in G_sim.nodes():
            current_state = G_sim.nodes[node]['state']
            new_states[node] = current_state

            if current_state == 'S':
                for neighbor in G_sim.neighbors(node):
                    if G_sim.nodes[neighbor]['state'] == 'I':
                        length = G_sim.edges[neighbor, node]['length']
                        adjusted_rate = base_propagation_rate / (1 + decay_factor * length)
                        if random.random() < adjusted_rate:
                            new_states[node] = 'I'
                            G_sim.nodes[node]['infected_time'] = t
                            break
            elif current_state == 'I':
                gamma = recovery_rate
                if intervention_active:
                    gamma += intervention_boost
                if random.random() < gamma:
                    new_states[node] = 'R'

        for node, state in new_states.items():
            G_sim.nodes[node]['state'] = state

        network_states_over_time.append({n: G_sim.nodes[n]['state'] for n in G_sim.nodes()})

        s, i, r = counts(G_sim)
        if s == 0:
            return t, network_states_over_time

    return num_time_steps, network_states_over_time

Q1：为什么可以借用传染病 SIR 模型来描述烟雾扩散？

参考方向：两者都是‘状态通过邻接关系转移’的过程：S→I 表示未受影响区域被影响；I→R 表示危险被清除。SIR 提供了一个成熟的状态机框架。

Q2：解释 base_propagation_rate、decay_factor、recovery_rate、intervention_boost 这四个参数的现实含义。

参考方向：base_propagation_rate：烟雾单位时间内的基础蔓延概率；decay_factor：通道越长，蔓延概率衰减越明显；recovery_rate：自然排烟/沉降使区域恢复安全的速率；intervention_boost：启动机械通风或喷淋后额外增加的恢复速率。

Q3（开放）：模型里的 R（Cleared）状态在现实中对应什么？它合理吗？

回答 A：对应烟雾被机械排烟系统抽出，或喷淋使烟尘沉降，区域重新达到可进入标准。

回答 B：对应消防员控制住火势并通风后，空间被宣布安全。

回答 C：在快速疏散场景中，R 可能不太合理，因为火灾现场很少在几分钟内‘恢复’；可以把 recovery_rate 设为 0，只看 S→I。

回答 D：R 更像是一种‘干预成功’的隐喻，提醒学生模型借用了传染病语言，需要检验每个状态的现实意义。

教师参考

本模块的解决方法是‘借用成熟模型框架并重新解释状态’。教师要让学生把 S、I、R 翻译成烟雾语言，而不是机械记忆。特别强调：base_propagation_rate 是概率不是速度；decay_factor 把几何长度引入传播；recovery_rate + intervention_boost 体现干预效果。

3. 基线模拟：没有干预时，烟多久充满商场？

为什么跑多次？因为传播是随机的，一次模拟可能只是‘运气好’或‘运气差’。我们需要蒙特卡洛方法：跑几百次，看平均行为和波动范围。

def run_smoke_simulation(graph, fire_origin, base_propagation_rate, decay_factor,
                         recovery_rate, intervention_boost, intervention_start):
    G_sim = graph.copy()
    for node in G_sim.nodes():
        G_sim.nodes[node]['state'] = 'S'
        G_sim.nodes[node]['infected_time'] = -1
    G_sim.nodes[fire_origin]['state'] = 'I'
    G_sim.nodes[fire_origin]['infected_time'] = 0
    intervention_active = False
    network_states_over_time = []

    for t in range(100):
        if t >= intervention_start and not intervention_active:
            if random.random() < 0.8:
                intervention_active = True
        new_states = {}
        for node in G_sim.nodes():
            current_state = G_sim.nodes[node]['state']
            new_states[node] = current_state
            if current_state == 'S':
                for neighbor in G_sim.neighbors(node):
                    if G_sim.nodes[neighbor]['state'] == 'I':
                        length = G_sim.edges[neighbor, node]['length']
                        rate = base_propagation_rate / (1 + decay_factor * length)
                        if random.random() < rate:
                            new_states[node] = 'I'
                            G_sim.nodes[node]['infected_time'] = t
                            break
            elif current_state == 'I':
                gamma = recovery_rate + (intervention_boost if intervention_active else 0)
                if random.random() < gamma:
                    new_states[node] = 'R'
        for node, state in new_states.items():
            G_sim.nodes[node]['state'] = state
        network_states_over_time.append({n: G_sim.nodes[n]['state'] for n in G_sim.nodes()})
        if sum(1 for n in G_sim.nodes() if G_sim.nodes[n]['state'] == 'S') == 0:
            return t, network_states_over_time
    return 100, network_states_over_time

4. 干预策略对比：隔离门 vs. 全楼通风

5. 网络度量与疏散路径：人往哪跑？

Q1：最短路径长度对疏散有什么实际意义？

参考方向：它表示从当前位置到安全出口‘最少需要经过的通道总长度’。在火灾中，这个长度越短，人员到达安全区域的时间通常越短。

Q2：介数中心性高的节点为什么特别关键？

参考方向：因为大量最短疏散路径都会经过它。如果它被封烟，很多人必须绕远路，疏散时间会显著增加。

Q3（开放）：如果某条走廊因烟不能通行，模型该如何反映人员的绕行？

回答 A：在图中临时删除该边，重新计算所有节点到出口的最短路径。

回答 B：把边的通行能力设为 0 或权重设得极大，让算法自动避开。

回答 C：引入时间维度：随着时间推移，某些边被烟‘关闭’，疏散路径动态变化。

回答 D：加入容量和拥堵模型：走廊能同时通过的人数有限，最短路径不一定最快。

目的：模型的输出（例如‘烟多久充满商场’）依赖于多个参数。敏感性分析帮助我们判断：如果某个参数估计不准，结论会不会大幅改变；以及应该把有限的资源投入到控制哪个因素上。

    ('Corridor_1', 'Stairwell_1')
])

    G_smoke.edges[u, v]['length'] = np.linalg.norm(np.array(pos[u]) - np.array(pos[v]))


def run_smoke_simulation(graph, fire_origin, base_propagation_rate, decay_factor,
                         recovery_rate, intervention_boost, intervention_start):
    G_sim = graph.copy()
    for node in G_sim.nodes():
        G_sim.nodes[node]['state'] = 'S'
    G_sim.nodes[fire_origin]['state'] = 'I'
    intervention_active = False
    for t in range(100):
        if t >= intervention_start and not intervention_active:
            if random.random() < 0.8:
                intervention_active = True
        new_states = {}
        for node in G_sim.nodes():
            state = G_sim.nodes[node]['state']
            new_states[node] = state
            if state == 'S':
                for neighbor in G_sim.neighbors(node):
                    if G_sim.nodes[neighbor]['state'] == 'I':
                        length = G_sim.edges[neighbor, node]['length']
                        rate = base_propagation_rate / (1 + decay_factor * length)
                        if random.random() < rate:
                            new_states[node] = 'I'
                            break
            elif state == 'I':
                gamma = recovery_rate + (intervention_boost if intervention_active else 0)
                if random.random() < gamma:
                    new_states[node] = 'R'
        for node, state in new_states.items():
            G_sim.nodes[node]['state'] = state
        if sum(1 for n in G_sim.nodes() if G_sim.nodes[n]['state'] == 'S') == 0:
            return t
    return 100


def avg_time(prop_rate, rec_rate, num_sims=50):
    times = []
    for _ in range(num_sims):
        t = run_smoke_simulation(
            G_smoke, 'Retail_A', prop_rate, 0.2, rec_rate, 0.0, 5
        )
        times.append(t)
    return np.mean(times)

import random
import numpy as np
import matplotlib.pyplot as plt
import networkx as nx

# Setup graph.
G_smoke = nx.Graph()
nodes_data = {
    'Retail_A': {'type': 'store', 'pos': (0, 0)},
    'Retail_B': {'type': 'store', 'pos': (1, 0)},
    'Retail_C': {'type': 'store', 'pos': (0, -1)},
    'Corridor_1': {'type': 'corridor', 'pos': (0.5, 0.5)},
    'Corridor_2': {'type': 'corridor', 'pos': (0.5, -0.5)},
    'Escalator_Up': {'type': 'escalator', 'pos': (1.5, 0.5)},
    'Escalator_Down': {'type': 'escalator', 'pos': (1.5, -0.5)},
    'Stairwell_1': {'type': 'stairwell', 'pos': (2, 0)},
    'Equipment_Room': {'type': 'utility', 'pos': (-0.5, -0.5)}
}
for node_id, data in nodes_data.items():
    G_smoke.add_node(node_id, type=data['type'], pos=data['pos'])
G_smoke.add_edges_from([
    ('Retail_A', 'Corridor_1'), ('Corridor_1', 'Retail_B'),
    ('Retail_A', 'Corridor_2'), ('Corridor_2', 'Retail_C'),
    ('Retail_B', 'Escalator_Up'), ('Escalator_Up', 'Stairwell_1'),
    ('Retail_C', 'Escalator_Down'), ('Escalator_Down', 'Stairwell_1'),
    ('Corridor_2', 'Equipment_Room'), ('Stairwell_1', 'Retail_B'),
    ('Corridor_1', 'Stairwell_1')
])
pos = nx.get_node_attributes(G_smoke, 'pos')
for u, v in G_smoke.edges():
    G_smoke.edges[u, v]['length'] = np.linalg.norm(np.array(pos[u]) - np.array(pos[v]))


def run_smoke_simulation(graph, fire_origin, base_propagation_rate=0.1,
                         decay_factor=0.2, recovery_rate=0.0):
    G_sim = graph.copy()
    for node in G_sim.nodes():
        G_sim.nodes[node]['state'] = 'S'
    G_sim.nodes[fire_origin]['state'] = 'I'
    states_over_time = []
    for t in range(100):
        new_states = {}
        for node in G_sim.nodes():
            state = G_sim.nodes[node]['state']
            new_states[node] = state
            if state == 'S':
                for neighbor in G_sim.neighbors(node):
                    if G_sim.nodes[neighbor]['state'] == 'I':
                        length = G_sim.edges[neighbor, node]['length']
                        rate = base_propagation_rate / (1 + decay_factor * length)
                        if random.random() < rate:
                            new_states[node] = 'I'
                            break
        for node, state in new_states.items():
            G_sim.nodes[node]['state'] = state
        states_over_time.append({n: G_sim.nodes[n]['state'] for n in G_sim.nodes()})
        if sum(1 for n in G_sim.nodes() if G_sim.nodes[n]['state'] == 'S') == 0:
            return states_over_time
    return states_over_time


store_nodes = [n for n, d in G_smoke.nodes(data=True) if d['type'] == 'store']
num_sims = 50

# Baseline: average affected stores over time.
baseline_counts = []
for _ in range(num_sims):
    states = run_smoke_simulation(G_smoke, 'Retail_A')
    counts = [sum(1 for n, s in st.items() if n in store_nodes and s == 'I') for st in states]
    baseline_counts.append(counts)
max_len = max(len(c) for c in baseline_counts)
baseline_padded = [c + [c[-1]] * (max_len - len(c)) for c in baseline_counts]
baseline_avg = np.array(baseline_padded).mean(axis=0)

# Test removing each edge.
edge_results = []
for u, v in G_smoke.edges():
    G_test = G_smoke.copy()
    G_test.remove_edge(u, v)
    counts_runs = []
    for _ in range(num_sims):
        states = run_smoke_simulation(G_test, 'Retail_A')
        counts = [sum(1 for n, s in st.items() if n in store_nodes and s == 'I') for st in states]
        counts_runs.append(counts)
    max_len = max(len(c) for c in counts_runs)
    padded = [c + [c[-1]] * (max_len - len(c)) for c in counts_runs]
    avg = np.array(padded).mean(axis=0)
    # Score: area under infected-store curve (smaller = better)
    score = avg.sum()
    edge_results.append(((u, v), score, avg))

edge_results.sort(key=lambda x: x[1])

print("Top 3 most effective single edges to block:")
for (u, v), score, _ in edge_results[:3]:
    print(f"  {u} - {v}: infected-store-time score = {score:.2f}")

# Plot top 2 vs baseline.
plt.figure(figsize=(10, 5))
plt.plot(baseline_avg, label='Baseline', color='blue')
for (u, v), _, avg in edge_results[:2]:
    plt.plot(avg, label=f'Block {u}-{v}')

plt.ylabel('Avg infected stores')
plt.title('Optimal Single Isolation Door Comparison')
plt.legend()
plt.grid(True, alpha=0.3)
